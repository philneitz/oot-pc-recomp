cmake_minimum_required(VERSION 3.22)
project(oot_mesh_xlate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LICHII_SOURCE "${REPO}/refs/lichii-mesh" CACHE PATH "Pinned Lichii checkout")
set(BOOST_HEADERS "${REPO}/build/mesh-deps/boost_1_89_0" CACHE PATH "Boost headers")
set(SPIRV_HEADERS "${REPO}/refs/mesh-spirv/include" CACHE PATH "SPIR-V include directory")
add_subdirectory("${REPO}/refs/mesh-fmt" "${CMAKE_BINARY_DIR}/fmt" EXCLUDE_FROM_ALL)
file(GLOB_RECURSE SIRIT_SRC "${REPO}/refs/mesh-sirit/src/*.cpp")
add_library(mesh_sirit STATIC ${SIRIT_SRC})
target_include_directories(mesh_sirit PUBLIC "${REPO}/refs/mesh-sirit/include" "${SPIRV_HEADERS}" PRIVATE "${REPO}/refs/mesh-sirit/src")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(mesh_sirit PRIVATE -include cstdlib)
endif()
file(GLOB_RECURSE RECOMP_SRC "${LICHII_SOURCE}/src/shader_recompiler/*.cpp")
list(FILTER RECOMP_SRC EXCLUDE REGEX "/backend/(glasm|glsl)/")
add_library(mesh_recompiler STATIC ${RECOMP_SRC} common_adapter.cpp)
target_include_directories(mesh_recompiler PUBLIC include "${LICHII_SOURCE}/src" "${BOOST_HEADERS}")
target_link_libraries(mesh_recompiler PUBLIC mesh_sirit fmt::fmt)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(mesh_recompiler PRIVATE -fbracket-depth=1024 -Wno-deprecated-declarations)
  target_compile_options(mesh_recompiler PUBLIC -Wno-pragma-clang-attribute)
endif()
add_executable(oot_mesh_xlate main.cpp)
target_link_libraries(oot_mesh_xlate PRIVATE mesh_recompiler)
if(MINGW)
  target_link_options(oot_mesh_xlate PRIVATE -static -static-libgcc -static-libstdc++)
endif()
